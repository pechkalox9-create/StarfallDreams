cmake_minimum_required(VERSION 3.20)
project(StarfallDreams VERSION 1.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(StarfallDreams WIN32 src/main.cpp)
if(WIN32)
  target_link_libraries(StarfallDreams PRIVATE shell32 ole32 comctl32 comdlg32)
endif()
