cmake_minimum_required(VERSION 3.21)
project(StarfallDreams LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
add_executable(StarfallDreams WIN32 src/main.cpp)
target_link_libraries(StarfallDreams PRIVATE Qt6::Widgets)
add_executable(DllTestHost src/test_host.cpp)
install(TARGETS StarfallDreams DllTestHost RUNTIME DESTINATION bin)
